using UnityEngine;
using UnityEngine.UI;
using System.Collections;

// TowerSpawner.cs
// Manages tower selection, placement, rotation, removal, and UI updates in a tower defense game.
// Ykada_Hiroka
public class TowerSpawner : MonoBehaviour
{
[Header("Preview Prefabs (Ghost Towers)")]
[SerializeField] private GameObject archerPreview;
[SerializeField] private GameObject crossbowPreview;
[SerializeField] private GameObject magePreview;
[SerializeField] private GameObject cannonPreview;
[SerializeField] private GameObject crystalPreview;
[SerializeField] private GameObject minigunnerPreview;

[Header("Tower Prefabs (Real Towers)")]
[SerializeField] private GameObject archer;
[SerializeField] private GameObject crossbow;
[SerializeField] private GameObject mage;
[SerializeField] private GameObject cannon;
[SerializeField] private GameObject crystal;
[SerializeField] private GameObject minigunner;

[Header("Settings")]
[SerializeField] private LayerMask placementLayer;
[SerializeField] private int maxTowers = 25;
[SerializeField] private int startingMoney = 500;
[SerializeField] private int money;
private float displayedMoney;

[Header("UI")]
[SerializeField] private Text moneyText;
[SerializeField] private Text towerErrorLog;
[SerializeField] private GameObject errorLogsPanel;
[SerializeField] private Text currenttowersspawnedtext;

private GameObject previewTower;
private GameObject selectedTowerPrefab;
private int previewCost;

private readonly List
	placedTowers = new List
		();

		private void Start()
		{
		money = startingMoney;
		displayedMoney = money;
		UpdateMoneyUI();
		}

		private void Update()
		{
		HandleTowerSelection();
		HandleTowerPlacement();
		HandleTowerInteraction();
		displayedMoney = Mathf.Lerp(displayedMoney, money, Time.deltaTime * 10f);
		moneyText.text = "$: " + Mathf.RoundToInt(displayedMoney);
		}

		#region Tower Selection
		private void HandleTowerSelection()
		{
		if (Input.GetKeyDown(KeyCode.Alpha1)) TrySelectTower(archer, archerPreview, 250);
		else if (Input.GetKeyDown(KeyCode.Alpha2)) TrySelectTower(crossbow, crossbowPreview, 500);
		else if (Input.GetKeyDown(KeyCode.Alpha3)) TrySelectTower(mage, magePreview, 750);
		else if (Input.GetKeyDown(KeyCode.Alpha4)) TrySelectTower(cannon, cannonPreview, 1500);
		else if (Input.GetKeyDown(KeyCode.Alpha5)) TrySelectTower(crystal, crystalPreview, 5000);
		else if (Input.GetKeyDown(KeyCode.Alpha6)) TrySelectTower(minigunner, minigunnerPreview, 15000);
		}

		private void TrySelectTower(GameObject prefab, GameObject previewPrefab, int cost)
		{
		if (placedTowers.Count >= maxTowers)
		{
		ShowError("Tower limit reached!");
		return;
		}

		if (money < cost)
    {
        ShowError("Insufficient funds!");
        return;
    }

    if (previewTower != null) Destroy(previewTower);

    previewTower = Instantiate(previewPrefab); // show ghost tower
    previewCost = cost;
    selectedTowerPrefab = prefab; // save real tower
}
#endregion

#region Tower Placement
private void HandleTowerPlacement()
{
    if (previewTower == null) return;

    Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
    if (Physics.Raycast(ray, out RaycastHit hit, 100f, placementLayer))
    {
        Vector3 newPos = hit.point;
        previewTower.transform.position = newPos;

        previewTower.transform.rotation = Quaternion.Euler(0f, previewTower.transform.rotation.eulerAngles.y, 0f);

        if (Input.GetKeyDown(KeyCode.R))
        {
            previewTower.transform.Rotate(0, 45f, 0);
        }

        if (Input.GetMouseButtonDown(0))
        {
            PlaceTower();
        }
    }
}

private void PlaceTower()
{
    if (money < previewCost)
    {
        ShowError("Not enough money!");
        Destroy(previewTower);
        previewTower = null;
        return;
    }

    money -= previewCost;
    UpdateMoneyUI();

    // Save preview position/rotation
    Vector3 pos = previewTower.transform.position;
    Quaternion rot = previewTower.transform.rotation;

    Destroy(previewTower); // remove preview

    // Spawn real tower
    GameObject realTower = Instantiate(selectedTowerPrefab, pos, rot);

    PlacedTower newTower = new PlacedTower(realTower, previewCost);
    placedTowers.Add(newTower);

    previewTower = null;
    selectedTowerPrefab = null;
}
#endregion

#region Tower Interaction (Rotate / Remove)
private void HandleTowerInteraction()
{
    if (previewTower != null) return;

    if (Input.GetKeyDown(KeyCode.R))
    {
        Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
        if (Physics.Raycast(ray, out RaycastHit hit))
        {
            PlacedTower tower = placedTowers.Find(t => t.TowerObject == hit.collider.gameObject);
            if (tower != null)
            {
                tower.TowerObject.transform.Rotate(0, 45f, 0);
            }
        }
    }

    if (Input.GetKeyDown(KeyCode.Backspace))
    {
        Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
        if (Physics.Raycast(ray, out RaycastHit hit))
        {
            PlacedTower tower = placedTowers.Find(t => t.TowerObject == hit.collider.gameObject);
            if (tower != null)
            {
                RemoveTower(tower);
            }
        }
    }
}

private void RemoveTower(PlacedTower tower)
{
    int refund = tower.Cost / 2;
    money += refund;
    UpdateMoneyUI();

    placedTowers.Remove(tower);
    Destroy(tower.TowerObject);
}
#endregion

#region Money & UI
public void AddMoney(int amount)
{
    money += amount;
    UpdateMoneyUI();
}

private void UpdateMoneyUI()
{

    moneyText.text = $"$ {money}";
    currenttowersspawnedtext.text = $"Towers: {placedTowers.Count}/{maxTowers}";
}

private void ShowError(string message)
{
    errorLogsPanel.SetActive(true);
    towerErrorLog.text = message;
    StartCoroutine(HideErrorAfterDelay(2.5f));
}

private IEnumerator HideErrorAfterDelay(float delay)
{
    yield return new WaitForSeconds(delay);
    errorLogsPanel.SetActive(false);
}
#endregion

#region Helper Class
private class PlacedTower
{
    public GameObject TowerObject { get; }
    public int Cost { get; }

    public PlacedTower(GameObject obj, int cost)
    {
        TowerObject = obj;
        Cost = cost;
    }
}
#endregion
}

[RequireComponent(typeof(Rigidbody))]
public class Tornado : MonoBehaviour
{
#region Private Fields

[Header("Tornado Settings (Loaded from Data Script)")]
private float TORNADOwindspeed;
private float TORNADObaseWindSpeed;
private float TORNADOCoreSpeed;
private float TORNADOgroundSpeed;
private float TORNADOtornadoRadius;
private float TORNADOliftForce;
private float TORNADOrotationSpeed;
private float TORNADOeffectRadius;
private float TORNADOmaxDistance;

private float TORNADOcurrentWindDirection;
private float TORNADOwindShiftSpeed;

[Header("Physics & Behavior")]
[SerializeField] private Rigidbody tornadoCore;
[SerializeField] private LayerMask interactableLayers;

[Header("Visuals & UI")]
[SerializeField] private ParticleSystem[] tornadoParticles;
[SerializeField] private Canvas tornadoUICanvas;
[SerializeField] private Text currentWindSpeedText;

[Header("System Integration")]
[SerializeField] private GeneralWetherSystem weatherSystem;
[SerializeField] private GeneralStormStatistics stats;
private bool DevelperOverWriteOfDebugSystem;

[Header("Debug & Player Tracking")]
[SerializeField] private Transform playerTransform;
private Vector3 rotationAxis = Vector3.up;
private Vector3 moveDirection;
private float turnSpeedPerSecond;

private float valueUpdateTimer = 0f;
[SerializeField] private float valueUpdateInterval = 10f;   // updates every 10 seconds

#endregion

private void Start()
{
turnSpeedPerSecond = 15f / 60f;
moveDirection = new Vector3(Random.Range(-1f, 1f), 0, Random.Range(-1f, 1f)).normalized;

if (tornadoCore == null)
tornadoCore = GetComponent
	();

	tornadoParticles = GetComponentsInChildren();
	DevelperOverWriteOfDebugSystem = stats.DevelperOverWriteOfDebugSystem;
	UpdateTornadoStats();   // initial load
	}
	private void Update()
	{
		// Update movement and physics every frame
		HandleWeatherBasedSpeed();
		HandlePlayerWindInteraction();

		//  2 second stat update 
		valueUpdateTimer += Time.deltaTime;
		if (valueUpdateTimer >= valueUpdateInterval)
		{
			valueUpdateTimer = 0f;
			UpdateTornadoStats();
		}

		// UI

	}
	private void FixedUpdate()
	{
		HandleMovement();
		//  HandleRotation();
	}
	#region Stats Update

	private void UpdateTornadoStats()
	{
		if (weatherSystem == null) return;

		// copy stats from weather system
		TORNADOwindspeed = stats.Windspeed;
		TORNADObaseWindSpeed = stats.BaseWindSpeed;
		TORNADOCoreSpeed = stats.MaxCoreSpeed;
		TORNADOgroundSpeed = stats.GroundSpeed;
		TORNADOtornadoRadius = stats.TornadoRadius;
		TORNADOliftForce = stats.LiftForce;
		TORNADOrotationSpeed = stats.RotationSpeed;
		TORNADOeffectRadius = stats.EffectRadius;
		TORNADOmaxDistance = stats.MaxDistance;
		//TORNADOwindShiftSpeed = stats.WindShiftSpeed;

		// copy wind direction from weather system
		TORNADOcurrentWindDirection = stats.CurrentWindDirection;

		// update UI
		currentWindSpeedText.text = $"Wind Speed: {TORNADOwindspeed} km/h";

		// debugs
		if (DevelperOverWriteOfDebugSystem == false)
		{
			Debug.Log($"[Tornado] Wind Speed: {TORNADOwindspeed}");
			Debug.Log($"[Tornado] Core Speed: {TORNADOCoreSpeed}");
			Debug.Log($"[Tornado] Ground Speed: {TORNADOgroundSpeed}");
			Debug.Log($"[Tornado] Tornado Radius: {TORNADOtornadoRadius}");
			Debug.Log($"[Tornado] Lift Force: {TORNADOliftForce}");
			Debug.Log($"[Tornado] Rotation Speed: {TORNADOrotationSpeed}");
		}
		else
		{ return; }

	}

	#endregion

	#region Tornado Movement
	private void HandleMovement()
	{
		if (tornadoCore == null) return;

		if (weatherSystem != null)
		{
			Vector3 targetDirection = weatherSystem.CurrentWindDirection.normalized;

			if (targetDirection.sqrMagnitude > 0.01f)
			{
				moveDirection = Vector3.RotateTowards(
					moveDirection,
					targetDirection,
					Mathf.Deg2Rad * turnSpeedPerSecond * Time.fixedDeltaTime,
					0f
				).normalized;
			}
		}

		Vector3 movement = moveDirection * TORNADOgroundSpeed * Time.fixedDeltaTime;
		tornadoCore.MovePosition(tornadoCore.position + movement);
	}


	private void HandleRotation()
	{
		transform.Rotate(rotationAxis, TORNADOrotationSpeed * Time.deltaTime, Space.World);

		if (tornadoCore != null)
			tornadoCore.AddTorque(rotationAxis * TORNADOrotationSpeed, ForceMode.Acceleration);
	}
	#endregion

	#region Weather & Speed
	private void HandleWeatherBasedSpeed()
	{
		TORNADOCoreSpeed = Mathf.Lerp(TORNADOCoreSpeed, TORNADObaseWindSpeed, Time.deltaTime * 0.5f);
	}
	#endregion

	#region Tornado Visuals

	//void UpdateTornadoVisuals()
	//{

	//    tornadoRadius = stats.tornadoRadius;
	    
	//    effectRadius = tornadoRadius * 2f;

	//}

	#endregion

	#region Player Interaction
	private void HandlePlayerWindInteraction()
	{
		if (playerTransform == null) return;

		float distance = Vector3.Distance(transform.position, playerTransform.position);
		float proximityFactor = Mathf.Clamp01(1 - (distance / TORNADOeffectRadius));

		float adjustedWind = Mathf.Lerp(TORNADObaseWindSpeed * 0.5f, TORNADObaseWindSpeed * 1.5f, proximityFactor);
		TORNADOCoreSpeed = adjustedWind;

		Collider[] nearby = Physics.OverlapSphere(transform.position, TORNADOeffectRadius, interactableLayers);
		foreach (var col in nearby)
		{
			Rigidbody rb = col.attachedRigidbody;
			if (rb != null)
			{
				Vector3 directionToCore = (transform.position - rb.position).normalized;
				float pullForce = TORNADOliftForce * proximityFactor;

				rb.AddForce(directionToCore * pullForce, ForceMode.Acceleration);
				rb.AddForce(Vector3.up * (TORNADOliftForce * 0.5f * proximityFactor), ForceMode.Acceleration);
			}
		}
	}
	#endregion
}

using UnityEngine;
using System.Collections;

public class OpenVualtDoor : MonoBehaviour
{
[Header("Door gameobjects")]
[SerializeField] private GameObject VualtDoor;
[SerializeField] private GameObject ParticalObjecs;
[SerializeField] private Transform player;

[Header("Door open settings")]
[SerializeField] private float openDoorSpeed = 2f;
[SerializeField] private float playerdetectionArea = 5f;
[SerializeField] private Vector3 openOffset;

private bool isDoorOpen = false;
private bool isMoving = false;

private Vector3 closedPosition;
private Vector3 openPosition;

private void Start()
{
closedPosition = VualtDoor.transform.position;
openPosition = closedPosition + openOffset;
}

private void Update()
{
float distanceToPlayer = Vector3.Distance(transform.position, player.position);

if (distanceToPlayer < playerdetectionArea && !isDoorOpen && !isMoving)
{
StartCoroutine(MoveDoor(openPosition));
isDoorOpen = true;
}
}

private IEnumerator MoveDoor(Vector3 targetPosition)
{
isMoving = true;
ParticalObjecs.SetActive(true);

while (Vector3.Distance(VualtDoor.transform.position, targetPosition) > 0.01f)
{
VualtDoor.transform.position = Vector3.MoveTowards(
VualtDoor.transform.position,
targetPosition,
openDoorSpeed * Time.deltaTime
);

yield return null;
}

ParticalObjecs.SetActive(false);
isMoving = false;
}
}
